function countOf(text, find) { return text.split(find).length - 1; }
function canon(value) {
  if (Array.isArray(value)) return value.map(canon);
  if (value && typeof value === 'object') return Object.fromEntries(Object.keys(value).sort().map((k) => [k, canon(value[k])]));
  return value;
}
function apiSeams(text) {
  const once = (find) => text.split(find).length - 1 === 1;
  if (!once('api.interceptors.request.use(') || !once('api.interceptors.response.use(') || !once('export function getApiErrorMessage(')) return null;
  const start = text.indexOf('api.interceptors.request.use(');
  const end = text.indexOf('\n);\n', text.indexOf('api.interceptors.response.use(')) + 4;
  const errStart = text.indexOf('export function getApiErrorMessage(');
  const errEnd = text.indexOf('\n}\n', errStart) + 3;
  return { interceptors: sha(text.slice(start, end)), errorMessage: sha(text.slice(errStart, errEnd)) };
}
function usersSeam(text) {
  const groups = [
    ['users table', /CREATE TABLE (?:IF NOT EXISTS )?"?users"?\s*\([\s\S]*?\);[ \t]*\n/g],
    ['users index', /CREATE UNIQUE INDEX [^;]*\bON\s+"?users"?\s*\([^;]*;/g],
    ['users triggers', /CREATE TRIGGER [^;]*?\bON\s+"?users"?\b[\s\S]*?\bEND;/g],
    ['sync_outbox table', /CREATE TABLE (?:IF NOT EXISTS )?"?sync_outbox"?\s*\([\s\S]*?\);[ \t]*\n/g],
    ['scoped_access_emit table', /CREATE TABLE (?:IF NOT EXISTS )?"?scoped_access_emit"?\s*\([\s\S]*?\);[ \t]*\n/g],
  ];
  const counts = {};
  const parts = [];
  for (const [label, re] of groups) {
    const found = text.match(re) || [];
    counts[label] = found.length;
    parts.push(...found);
  }
  return { sha256: sha(parts.join('\n--\n')), counts };
}

// .json, .node, directory package.json main, directory index.js. An explicit
// .cjs name resolves as a file.
function isFile(rel) { try { return fs.statSync(rel).isFile(); } catch (_) { return false; } }
function resolveLocal(fromRel, spec) {
  const base = path.posix.normalize(path.posix.join(path.posix.dirname(fromRel), spec));
  for (const cand of [base, base + '.js', base + '.json', base + '.node']) if (isFile(cand)) return [cand];
  const out = [];
  if (isFile(base + '/package.json')) {
    out.push(base + '/package.json');
    const main = JSON.parse(fs.readFileSync(base + '/package.json', 'utf8')).main;
    if (main) {
      const entry = path.posix.normalize(path.posix.join(base, main));
      for (const cand of [entry, entry + '.js', entry + '/index.js']) if (isFile(cand)) { out.push(cand); return out; }
    }
  }
  if (isFile(base + '/index.js')) { out.push(base + '/index.js'); return out; }
  return null;
}
function requireClosure(roots, failures) {
  const seen = new Set();
  const stack = [...roots];
  while (stack.length) {
    const rel = stack.pop();
    if (seen.has(rel)) continue;
    seen.add(rel);
    if (!isFile(rel)) { failures.push(`closure input missing: ${rel}`); continue; }
    if (!/\.c?js$/.test(rel)) continue;
    const text = fs.readFileSync(rel, 'utf8');
    for (const m of text.matchAll(/require(?:\.resolve)?\(\s*(['"])(\.{1,2}\/[^'"]+)\1\s*\)/g)) {
      const resolved = resolveLocal(rel, m[2]);
      if (!resolved) failures.push(`unresolved dependency: ${rel} -> ${m[2]}`);
      else stack.push(...resolved);
    }
  }
  return [...seen].sort();
}

// Git tree rule. Entries come from Git objects only, never from the checkout.
function lsTree(commit) {
  const out = git(['ls-tree', '-r', '-z', '--full-tree', commit]);
  const map = new Map();
  for (const rec of out.split('\0').filter(Boolean)) {
    const tab = rec.indexOf('\t');
    const [mode, type, oid] = rec.slice(0, tab).split(' ');
    map.set(rec.slice(tab + 1), { mode, type, oid });
  }
  return map;
}
function rawDiff(from, to) {
  const parts = git(['diff-tree', '-r', '-z', '--raw', '--no-renames', '--full-index', from, to]).split('\0').filter(Boolean);
  const entries = [];
  for (let i = 0; i < parts.length; i += 2) {
    const [oldMode, newMode, oldOid, newOid, status] = parts[i].slice(1).split(' ');
    entries.push({ status, path: parts[i + 1], oldMode, newMode, oldOid, newOid });
  }
  return entries;
}
function treeRecord(base, classAKeys, failures) {
  let reviewed = [];
  try { reviewed = rawDiff(M, N); } catch (_) { failures.push(`cannot read ${N}; fetch main first`); }
  if (reviewed.length !== REVIEWED_MAIN_DIFF.entries || sha(JSON.stringify(reviewed)) !== REVIEWED_MAIN_DIFF.sha256) {
    failures.push('the reviewed tree diff M -> N does not reproduce');
  }
  const reviewedKeys = new Set(reviewed.map((e) => JSON.stringify(e)));
  const baseTree = lsTree(base);
  const gitlinks = [...baseTree.entries()].filter(([, e]) => e.mode === '160000').map(([p, e]) => ({ path: p, oid: e.oid }));
  if (gitlinks.length) failures.push(`the base tree has gitlinks, whose content the record cannot bind: ${gitlinks.map((g) => g.path).join(', ')}`);
  const sPaths = new Set([...Object.keys(CLASS_S_FILES), ...Object.keys(CLASS_S_LINKS)]);
  const diff = rawDiff(M, base).map((e) => ({
    ...e,
    cls: sPaths.has(e.path) ? 'S' : classAKeys.has(e.path) ? 'named' : 'tree',
    reviewedAtN: reviewedKeys.has(JSON.stringify(e)),
  }));
  for (const e of diff) if (e.cls === 'S') failures.push(`class S tree entry changed: ${e.path} (${e.status} ${e.oldMode}->${e.newMode})`);
  const mTree = lsTree(M);
  const modes = {};
  for (const rel of sPaths) {
    const atM = mTree.get(rel);
    const atBase = baseTree.get(rel);
    modes[rel] = atBase ? atBase.mode : null;
    if (!atM || !atBase || atM.mode !== atBase.mode || atM.oid !== atBase.oid) failures.push(`class S entry differs from M in the base tree: ${rel}`);
  }
  const resolution = {};
  for (const dir of RESOLUTION_GUARDS) {
    const shadows = ['.js', '.json', '.node'].map((ext) => dir + ext).filter((p) => baseTree.has(p) || fs.existsSync(p));
    resolution[dir] = shadows;
    if (shadows.length) failures.push(`a file shadows the directory load of ${dir}: ${shadows.join(', ')}`);
  }
  return {
    root: git(['rev-parse', `${base}^{tree}`]).trim(),
    rootAtM: git(['rev-parse', `${M}^{tree}`]).trim(),
    reviewedMain: { commit: N, ...REVIEWED_MAIN_DIFF },
    gitlinks,
    classSModes: modes,
    resolution,
    diff,
    needsReview: diff.filter((e) => !e.reviewedAtN).map((e) => e.path),
  };
}
// Ignored paths can change module or tool resolution (an ignored
// node_modules directory, for example) but are not in any Git tree; they are
// listed, collapsed per directory as git status prints them.
function ignoredPaths() {
  return git(['status', '--porcelain=v1', '-z', '--ignored=traditional', '--untracked-files=normal']).split('\0')
    .filter((rec) => rec.startsWith('!! ')).map((rec) => rec.slice(3)).sort();
}

// The record describes the checked-out tree and the base commit's Git tree.
// It is written at the base and recomputed on the plan commit; both must be
// identical apart from `base`.
function computeRecord(base) {
  const failures = [];
  const drift = [];
  const record = { M, N, custodySha256: SELF_SHA256, classS: {}, classSLinks: {}, classA: {}, anchors: {} };
  for (const [rel, expected] of Object.entries(CLASS_S_FILES)) {
    const actual = isFile(rel) ? sha(fs.readFileSync(rel)) : null;
    record.classS[rel] = { expected, actual };
    if (actual !== expected) failures.push(`class S file changed: ${rel} (${actual})`);
  }
  for (const [rel, expected] of Object.entries(CLASS_S_LINKS)) {
    let actual = null;
    try { if (fs.lstatSync(rel).isSymbolicLink()) actual = fs.readlinkSync(rel); } catch (_) { actual = null; }
    record.classSLinks[rel] = { expected, actual };
    if (actual !== expected) failures.push(`class S link changed: ${rel} (${actual})`);
  }
  const raws = PROFILE_ROOTS.map((r) => fs.readFileSync(`${r}/flows.json`, 'utf8'));
  if (raws[0] !== raws[1]) failures.push('the two maintained flows.json profiles differ');
  raws.forEach((raw, i) => {
    const flows = JSON.parse(raw);
    if (JSON.stringify(flows, null, 2) + '\n' !== raw) failures.push(`${PROFILE_ROOTS[i]}/flows.json does not round-trip`);
    const byId = new Map(flows.map((n) => [n.id, n]));
    for (const [id, expected] of Object.entries(CLASS_S_NODES)) {
      const actual = byId.has(id) ? sha(JSON.stringify(byId.get(id))) : null;
      record.classS[`${PROFILE_ROOTS[i]}/flows.json#${id}`] = { expected, actual };
      if (actual !== expected) failures.push(`class S node changed: ${id} in ${PROFILE_ROOTS[i]} (${actual})`);
    }
    const atM = gitShow(`${PROFILE_ROOTS[i]}/flows.json`);
    if (atM) {
      const old = new Map(JSON.parse(atM.toString('utf8')).map((n) => [n.id, JSON.stringify(n)]));
      const changed = flows.filter((n) => old.get(n.id) !== JSON.stringify(n)).map((n) => ({ id: n.id, sha256: sha(JSON.stringify(n)) }));
      const removed = [...old.keys()].filter((id) => !byId.has(id));
      record.anchors[`${PROFILE_ROOTS[i]}/flows.json nodes changed since M`] = { changed, removed };
    } else {
      failures.push(`cannot read ${PROFILE_ROOTS[i]}/flows.json at ${M}; fetch main first`);
    }
  });
  const workflowDir = fs.readdirSync('.github/workflows').filter((f) => /\.ya?ml$/.test(f)).sort().map((f) => `.github/workflows/${f}`);
  record.anchors.workflows = workflowDir;
  const closure = requireClosure(CLOSURE_ROOTS, failures);
  record.anchors.closure = closure;
  const classA = new Set([...Object.keys(CLASS_A_FILES), ...CLASS_A_REPORT, ...workflowDir, ...EXEC_INPUTS, ...GUI_INPUTS,
    ...closure.filter((rel) => !Object.prototype.hasOwnProperty.call(CLASS_S_FILES, rel))]);
  for (const rel of [...classA].sort()) {
    const actual = isFile(rel) ? sha(fs.readFileSync(rel)) : null;
    const shown = Object.prototype.hasOwnProperty.call(CLASS_A_FILES, rel) ? null : gitShow(rel);
    const atM = CLASS_A_FILES[rel] || (shown ? sha(shown) : null);
    record.classA[rel] = { atM, actual };
    if (actual === null) failures.push(`class A input missing: ${rel}`);
    else if (actual !== atM) drift.push(rel);
  }
  // Anchors the editors and gates rely on.
  const seam = usersSeam(fs.readFileSync('database/seed-blank.sql', 'utf8'));
  record.anchors.usersSeam = seam;
  if (seam.sha256 !== USERS_SEAM_SHA256) failures.push(`seed users seam changed (${seam.sha256}): composition path`);
  const api = fs.readFileSync('web/react-gui/src/services/api.ts', 'utf8');
  const seams = apiSeams(api);
  record.anchors.apiSeams = seams;
  if (!seams || seams.interceptors !== API_SEAMS.interceptors || seams.errorMessage !== API_SEAMS.errorMessage) failures.push('api.ts session interceptors or getApiErrorMessage changed');
  for (const anchor of ["export const listUsers = () =>\n  api.get<{ users: AdminUser[] }>('/api/users')", 'export const resetPassword = (uuid: string, password: string) =>\n', '  AdminUser,\n  WatermarkCalibrationState,']) {
    if (countOf(api, anchor) !== 1) failures.push(`api.ts anchor not found exactly once: ${anchor.slice(0, 40)}`);
  }
  if (countOf(fs.readFileSync('web/react-gui/src/types/farming.ts', 'utf8'), 'export interface AdminUser {\n') !== 1) failures.push('farming.ts AdminUser anchor not found exactly once');
  const allowanceRaw = fs.readFileSync('scripts/verify-flows-size-ratchet-allowances.json', 'utf8');
  const allowances = JSON.parse(allowanceRaw);
  if (JSON.stringify(allowances, null, 2) + '\n' !== allowanceRaw) failures.push('allowances do not round-trip');
  if (allowances.node_allowances['scoped-admin-account-router']) failures.push('a router node allowance already exists: composition path');
  const delta = allowances.total_allowance && allowances.total_allowance.delta;
  record.anchors.totalDelta = delta;
  const identity = fs.readFileSync('scripts/verify-live-gateway-identity.js', 'utf8');
  if (countOf(identity, `  expectCondition(sizeAllowances.total_allowance?.delta === ${delta},\n`) !== 1) failures.push(`identity pin for total delta ${delta} not found exactly once`);
  // Two reviewed label forms: main's "exact cumulative delta" and the
  // "exact delta" block written by the #356 R4 editor.
  const labelForms = {
    cumulative: `    'size total allowance: exact cumulative delta ${delta}',\n    'size total allowance: expected exact cumulative delta ${delta}');\n`,
    plain: `    'size total allowance: exact delta ${delta}',\n    'size total allowance: expected exact delta ${delta}');\n`,
  };
  const labelCounts = Object.fromEntries(Object.entries(labelForms).map(([k, v]) => [k, countOf(identity, v)]));
  const forms = Object.keys(labelCounts).filter((k) => labelCounts[k] > 0);
  record.anchors.identityLabels = { counts: labelCounts, form: forms.length === 1 && labelCounts[forms[0]] === 1 ? forms[0] : null };
  if (!record.anchors.identityLabels.form) failures.push(`identity labels for total delta ${delta}: expected exactly one reviewed form once (${JSON.stringify(labelCounts)})`);
  const workflow = fs.readFileSync('.github/workflows/migrations.yml', 'utf8');
  if (countOf(workflow, CI_ANCHOR) !== 1) failures.push('migrations.yml scoped-access write step anchor not found exactly once');
  record.anchors.ciNeighbourSteps = workflow.split('\n').filter((l) => /- name: .*(issue 443|#443|Registration zone authority)/.test(l));
  const typecheck = fs.readFileSync('.github/workflows/typecheck.yml', 'utf8');
  for (const line of ['run: npm ci', 'run: npm run typecheck', 'run: npm run test:unit']) if (!typecheck.includes(line)) failures.push(`typecheck.yml lacks "${line}"`);
  const pkg = JSON.parse(fs.readFileSync('web/react-gui/package.json', 'utf8'));
  if (pkg.scripts.typecheck !== 'tsc --noEmit' || pkg.scripts.build !== 'vite build') failures.push('GUI typecheck or build script changed');
  for (const dir of ['src/pages/__tests__', 'src/services/__tests__']) if (!String(pkg.scripts['test:unit:vitest']).includes(dir)) failures.push(`test:unit:vitest no longer collects ${dir}`);
  for (const locale of ['en', 'de-CH', 'es', 'fr', 'it', 'lg', 'pt']) {
    const rel = `web/react-gui/public/locales/${locale}/settings.json`;
    const raw = fs.readFileSync(rel, 'utf8');
    const parsed = JSON.parse(raw);
    const keys = Object.keys(parsed);
    if (JSON.stringify(parsed, null, 2) + '\n' !== raw) failures.push(`${rel} does not round-trip`);
    if (Object.prototype.hasOwnProperty.call(parsed, 'users')) failures.push(`${rel} already has users`);
    if (keys[keys.length - 1] !== 'grants') failures.push(`${rel}: grants is not the last key`);
  }
  const tracker = fs.readFileSync('docs/i18n/pending-luganda-translations.md', 'utf8');
  if (!tracker.startsWith('# Pending human Luganda translations\n') || !tracker.endsWith('\n') || tracker.endsWith('\n\n')) failures.push('Luganda tracker header or ending changed');
  if (tracker.includes('settings.json` — Users page')) failures.push('Luganda tracker already has the Users section');
  for (const { dest } of Object.values(PAYLOADS)) if (dest && fs.existsSync(dest)) failures.push(`test destination already exists: ${dest}`);
  record.tree = treeRecord(base, classA, failures);
  record.anchors.ignored = ignoredPaths();
  record.drift = drift;
  record.failures = failures;
  return record;
}
function withoutBase(record) { const { base: _b, ...rest } = record; return JSON.stringify(canon(rest)); }

function verifyInputs(recordPath) {
  checkRoot();
  requireClean();
  const head = headCommit();
  const record = { base: head, ...computeRecord(head) };
  if (recordPath) {
    if (insideRepo(recordPath)) die('write the record to scratch; materialize copies it into the plan');
    fs.writeFileSync(recordPath, JSON.stringify(record, null, 2) + '\n');
  }
  console.log(JSON.stringify({
    base: record.base, custody: SELF_SHA256, drift: record.drift, failures: record.failures, totalDelta: record.anchors.totalDelta,
    identityLabels: record.anchors.identityLabels.form, workflows: record.anchors.workflows.length, closure: record.anchors.closure.length,
    classA: Object.keys(record.classA).length, treeDiff: record.tree.diff.length,
    reviewedAtN: record.tree.diff.filter((e) => e.reviewedAtN).length, needsReview: record.tree.needsReview, ignored: record.anchors.ignored,
    flowNodesChangedSinceM: Object.values(record.anchors).filter((v) => v && v.changed).map((v) => v.changed.map((c) => c.id)),
  }, null, 2));
  if (record.failures.length) process.exit(1);
  const moved = record.drift.length || record.tree.diff.length;
  console.log(moved ? 'custody-521: drift from M without class-S change; review the record with the plan commit' : 'custody-521: inputs match the reviewed base');
  process.exit(moved ? 3 : 0);
}

function copyVerified(from, to, expected) {
  const bytes = fs.readFileSync(from);
  if (sha(bytes) !== expected) die(`${from} is not the reviewed bytes (${sha(bytes)})`);
  if (fs.existsSync(to)) die(`refusing to overwrite ${to}`);
  fs.mkdirSync(path.dirname(to), { recursive: true });
  fs.writeFileSync(to, bytes);
  if (sha(fs.readFileSync(to)) !== expected) die(`${to} differs after the write`);
}

function readRecord(recordFile) {
  const record = JSON.parse(fs.readFileSync(recordFile, 'utf8'));
  if (record.M !== M || record.N !== N || record.custodySha256 !== SELF_SHA256 || !Array.isArray(record.failures) || record.failures.length ||
      !record.tree || !Array.isArray(record.tree.diff) || !/^[0-9a-f]{40}$/.test(String(record.base))) {
    die('the reconciliation record is not a passing record written by this checker');
  }
  return record;
}

function materialize(packetRoot, recordFile, planSha256) {
  checkRoot();
  if (!packetRoot || !recordFile || !/^[0-9a-f]{64}$/.test(String(planSha256))) die('usage: materialize <packetRoot> <recordFile> <planSha256>');
  requireClean();
  const record = readRecord(recordFile);
  if (record.base !== headCommit()) die(`the record was written at ${record.base}, not at HEAD`);
  if (withoutBase(computeRecord(record.base)) !== withoutBase(record)) die('the inputs changed after the record was written');
  if (fs.existsSync(PLAN_DOC) || fs.existsSync(SUPPORT_DIR)) die('the plan is already materialized');
  const planBytes = fs.readFileSync(path.join(packetRoot, PLAN_DOC));
  if (sha(planBytes) !== planSha256) die(`the plan document is not the reviewed bytes (${sha(planBytes)})`);
  const selfBytes = fs.readFileSync(path.join(packetRoot, SUPPORT_DIR, 'custody-521.js.txt'));
  if (sha(selfBytes) !== SELF_SHA256) die('the packet custody checker differs from the running copy');
  fs.mkdirSync(SUPPORT_DIR, { recursive: true });
  fs.writeFileSync(PLAN_DOC, planBytes);
  for (const [name, { sha256 }] of Object.entries(PAYLOADS)) {
    copyVerified(path.join(packetRoot, SUPPORT_DIR, `${name}.txt`), path.join(SUPPORT_DIR, `${name}.txt`), sha256);
  }
  fs.writeFileSync(path.join(SUPPORT_DIR, 'custody-521.js.txt'), selfBytes);
  fs.writeFileSync(path.join(SUPPORT_DIR, 'reconciliation.json'), JSON.stringify(record, null, 2) + '\n');
  console.log(JSON.stringify({ base: record.base, plan: sha(planBytes), custody: SELF_SHA256, reconciliation: sha(fs.readFileSync(path.join(SUPPORT_DIR, 'reconciliation.json'))), paths: PLAN_PATHS.length }, null, 2));
}

function recheck(recordFile, planCommit) {
  checkRoot();
  if (!recordFile || !/^[0-9a-f]{40}$/.test(String(planCommit))) die('usage: recheck <committedRecord> <planCommit>');
  requireClean();
  const head = headCommit();
  if (head !== planCommit) die(`HEAD ${head} is not the reviewed plan commit ${planCommit}`);
  const record = readRecord(recordFile);
  const parents = git(['rev-list', '--parents', '-n', '1', 'HEAD']).trim().split(/\s+/).slice(1);
  if (parents.length !== 1 || parents[0] !== record.base) die(`the plan commit's parent ${parents.join(',')} is not the recorded base ${record.base}`);
  // Exactly the plan files, each added as a regular file (mode 100644).
  const added = rawDiff(record.base, 'HEAD').map((e) => `${e.status} ${e.oldMode} ${e.newMode}\t${e.path}`);
  const expected = PLAN_PATHS.map((rel) => `A 000000 100644\t${rel}`);
  if (JSON.stringify(added.slice().sort()) !== JSON.stringify(expected.slice().sort())) die('the plan commit does not add exactly the plan files:\n' + added.join('\n'));
  // The record must describe the Git objects at the base, not only this tree.
  for (const [group, entries] of [['classS', record.classS], ['classA', record.classA]]) {
    for (const [key, { actual }] of Object.entries(entries)) {
      if (key.includes('#')) continue;
      const shown = gitShowAt(record.base, key);
      if ((shown ? sha(shown) : null) !== actual) die(`${group} ${key}: the record does not match the base object`);
    }
  }
  for (const [key, { actual }] of Object.entries(record.classSLinks)) {
    const shown = gitShowAt(record.base, key);
    if ((shown ? shown.toString('utf8') : null) !== actual) die(`classSLinks ${key}: the record does not match the base object`);
  }
  const fresh = computeRecord(record.base);
  if (fresh.failures.length) die('inputs now fail: ' + fresh.failures.join('; '));
  if (withoutBase(fresh) !== withoutBase(record)) {
    const changed = [];
    if (JSON.stringify(canon(fresh.tree)) !== JSON.stringify(canon(record.tree))) changed.push('tree');
    for (const group of ['classS', 'classSLinks', 'classA']) {
      const keys = new Set([...Object.keys(fresh[group]), ...Object.keys(record[group])]);
      for (const key of keys) if (JSON.stringify(fresh[group][key]) !== JSON.stringify(record[group][key])) changed.push(`${group} ${key}`);
    }
    for (const key of new Set([...Object.keys(fresh.anchors), ...Object.keys(record.anchors)])) {
      if (JSON.stringify(canon(fresh.anchors[key])) !== JSON.stringify(canon(record.anchors[key]))) changed.push(`anchor ${key}`);
    }
    die('recorded inputs changed since the reviewed reconciliation: ' + (changed.join('; ') || 'drift list'));
  }
  console.log(`custody-521: plan commit ${head} on base ${record.base}: every recorded input unchanged`);
}

function extract(fromDir, toDir) {
  checkRoot();
  if (!fromDir || !toDir) die('usage: extract <fromDir> <toDir>');
  if (insideRepo(toDir)) die('extract into scratch outside the repository');
  for (const [name, { sha256 }] of Object.entries(PAYLOADS)) copyVerified(path.join(fromDir, `${name}.txt`), path.join(toDir, name), sha256);
  console.log(`custody-521: ${Object.keys(PAYLOADS).length} payloads extracted to ${toDir}`);
}

function stageTests(scratchDir) {
  checkRoot();
  if (!scratchDir) die('usage: stage-tests <scratchDir>');
  for (const [name, { sha256, dest }] of Object.entries(PAYLOADS)) if (dest) copyVerified(path.join(scratchDir, name), dest, sha256);
  console.log('custody-521: five tests staged');
}

const [command, ...rest] = process.argv.slice(2);
if (command === 'verify-inputs') verifyInputs(rest[0] === '--record' ? rest[1] : null);
else if (command === 'materialize') materialize(rest[0], rest[1], rest[2]);
else if (command === 'recheck') recheck(rest[0], rest[1]);
else if (command === 'extract') extract(rest[0], rest[1]);
else if (command === 'stage-tests') stageTests(rest[0]);
else if (command === 'seam-digest') console.log(JSON.stringify(usersSeam(fs.readFileSync(rest[0], 'utf8'))));
else die('usage: custody-521.js verify-inputs [--record <file>] | materialize <packetRoot> <record> <planSha256> | recheck <record> <planCommit> | extract <fromDir> <toDir> | stage-tests <scratchDir> | seam-digest <seed.sql>');
