#!/usr/bin/env node
'use strict';
// #521 GUI, locale, Luganda-tracking and CI editor (plan R2). Runs from a
// scratch copy outside the repository, with cwd at the root of the admitted
// implementation worktree:
//   node <scratch>/edit-gui-521.js <scratch>/UsersPage.candidate.tsx            dry run
//   node <scratch>/edit-gui-521.js <scratch>/UsersPage.candidate.tsx --write    applies every edit
// Every trusted digest is a constant in this reviewed file; nothing is read
// from the environment or from a caller-supplied summary. The three GUI
// sources must carry their reviewed bytes, every anchor must occur exactly
// once and every locale file must round-trip.
const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');

const sha = (text) => crypto.createHash('sha256').update(text).digest('hex');
function fail(message) { console.error('edit-gui-521: FAIL - ' + message); process.exit(1); }

if (!fs.existsSync('.git') || !fs.existsSync('web/react-gui/package.json')) fail('run from the root of the implementation worktree');
if (path.resolve(__dirname).startsWith(path.resolve('.') + path.sep)) {
  fail('run the scratch copy outside the repository');
}

// UsersPage.tsx is replaced whole, so it must carry the reviewed bytes
// (identical at f6682ef6 and 74ff869e). farming.ts and api.ts may carry
// unrelated hunks from other work: their edit anchors must occur exactly once,
// and in api.ts the session interceptors and getApiErrorMessage, which the
// new error typing relies on, must keep their reviewed digests.
const FILE_PINS = {
  'web/react-gui/src/pages/admin/UsersPage.tsx': '0b8e7833b7c9b1108a9be0fd870ff18e1004a70b649521e8fbac0dd101e9c6fd',
};
const API_SEAMS = { interceptors: 'b0a18d1a6cb8b1c164cc8e658c5c360b1f424024de5ef5e97037d7e69beb7377', errorMessage: '4246ce7796e0996b69c7c9638ef0042355abddb8045e2795c267288690e2bccd' };
function apiSeams(text) {
  const once = (find) => text.split(find).length - 1 === 1;
  if (!once('api.interceptors.request.use(') || !once('api.interceptors.response.use(') || !once('export function getApiErrorMessage(')) return null;
  const start = text.indexOf('api.interceptors.request.use(');
  const end = text.indexOf('\n);\n', text.indexOf('api.interceptors.response.use(')) + 4;
  const errStart = text.indexOf('export function getApiErrorMessage(');
  const errEnd = text.indexOf('\n}\n', errStart) + 3;
  return { interceptors: sha(text.slice(start, end)), errorMessage: sha(text.slice(errStart, errEnd)) };
}
const USERS_PAGE_CANDIDATE_SHA256 = '1b1c9976faf53230275ed5dc828b1bf149072a9a847ab20251eafc6f1381be37';

function replaceOnce(text, find, replace, label) {
  const count = text.split(find).length - 1;
  if (count !== 1) fail(`${label}: anchor occurs ${count} times`);
  return text.replace(find, () => replace);
}

const edits = new Map();
function pinned(rel) {
  const text = fs.readFileSync(rel, 'utf8');
  if (sha(text) !== FILE_PINS[rel]) fail(`${rel} is not the reviewed bytes (${sha(text)}); revise the plan`);
  return text;
}

// 1. Domain type.
{
  const rel = 'web/react-gui/src/types/farming.ts';
  edits.set(rel, replaceOnce(fs.readFileSync(rel, 'utf8'),
    'export interface AdminUser {\n  username: string;\n  user_uuid: string;\n  role: \'admin\' | \'researcher\' | \'viewer\';\n  disabled_at: string | null;\n  created_at: string;\n}\n',

    "export type AdminAuthMode = 'local' | 'server' | 'unknown';\n\n" +

    '  /** Normalized in services/api.ts; rows from an older gateway read as unknown. */\n  auth_mode: AdminAuthMode;\n}\n',
    rel));
}

{

    '/** The reset route\'s code for an account that signs in through its cloud link (#521). */\n' +
    "export const LINKED_PASSWORD_RESET_UNSUPPORTED = 'linked_account_password_reset_unsupported';\n\n" +
    '/** A local password reset the gateway refused because the account uses linked sign-in. */\n' +
    'export class LinkedPasswordResetError extends Error {\n' +
    '  readonly code = LINKED_PASSWORD_RESET_UNSUPPORTED;\n\n' +
    '  constructor(message: string) {\n    super(message);\n    this.name = \'LinkedPasswordResetError\';\n  }\n}\n\n' +
    'export function isLinkedPasswordResetError(error: unknown): error is LinkedPasswordResetError {\n' +
    '  return error instanceof LinkedPasswordResetError;\n}\n\n' +
    '// An older gateway sends no auth_mode. Anything other than local or server\n' +
    '// reads as unknown, so the page never offers a reset it cannot vouch for.\n' +
    'export function normaliseAdminAuthMode(value: unknown): AdminAuthMode {\n' +
    "  return value === 'local' || value === 'server' ? value : 'unknown';\n}\n\n" +
    'function toAdminUser(row: unknown): AdminUser {\n' +
    "  const raw = (row && typeof row === 'object' ? row : {}) as Partial<AdminUser> & { auth_mode?: unknown };\n" +
    '  return { ...(raw as AdminUser), auth_mode: normaliseAdminAuthMode(raw.auth_mode) };\n}\n\n' +
    'export const listUsers = () =>\n' +
    "  api.get<{ users?: unknown }>('/api/users').then((response) => {\n" +
    '    const users = response.data?.users;\n' +
    "    if (!Array.isArray(users)) throw new Error('The gateway returned an unreadable user list.');\n" +
    '    return users.map(toAdminUser);\n  });\n\n' +
    "export const createUser = (body: { username: string; password: string; role: AdminUser['role'] }) =>\n" +
    "  api.post<unknown>('/api/users', body).then((response) => toAdminUser(response.data));\n\n" +
    '// Only HTTP 409 with the exact code is the linked refusal; every other\n' +
    '// failure, including a 401 the session interceptor already handled, passes\n' +
    '// through unchanged.\n' +
    'export const resetPassword = (uuid: string, password: string): Promise<void> =>\n' +
    '  api.post(`/api/users/${encodeURIComponent(uuid)}/password-reset`, { password }).then(\n' +
    '    () => undefined,\n' +
    '    (error: unknown) => {\n' +
    '      if (axios.isAxiosError<{ code?: unknown }>(error) && error.response?.status === 409\n' +
    '          && error.response.data?.code === LINKED_PASSWORD_RESET_UNSUPPORTED) {\n' +
    "        throw new LinkedPasswordResetError(getApiErrorMessage(error, 'This account uses linked sign-in.'));\n" +
    '      }\n      throw error;\n    },\n  );\n',
    rel + ' users block');
  edits.set(rel, text);
}

// 3. Page.
{
  const rel = 'web/react-gui/src/pages/admin/UsersPage.tsx';
  pinned(rel);
  const candidatePath = process.argv.slice(2).find((a) => !a.startsWith('--'));
  if (!candidatePath) fail('usage: edit-gui-521.js <UsersPage.candidate.tsx> [--write]');
  const candidate = fs.readFileSync(candidatePath, 'utf8');
  if (sha(candidate) !== USERS_PAGE_CANDIDATE_SHA256) {
    fail(`UsersPage candidate is not the reviewed bytes (${sha(candidate)})`);
  }
  edits.set(rel, candidate);
}

for (const locale of ['en', 'de-CH', 'es', 'fr', 'it', 'lg', 'pt']) {
  const rel = `web/react-gui/public/locales/${locale}/settings.json`;
  const raw = fs.readFileSync(rel, 'utf8');
  const parsed = JSON.parse(raw);
  if (JSON.stringify(parsed, null, 2) + '\n' !== raw) fail(`${rel} does not round-trip; stop`);
  if (Object.prototype.hasOwnProperty.call(parsed, 'users')) fail(`${rel} already has users`);
  const keys = Object.keys(parsed);
  if (keys[keys.length - 1] !== 'grants') fail(`${rel}: grants is not the last key`);
  edits.set(rel, JSON.stringify({ ...parsed, users: USERS[locale] }, null, 2) + '\n');
}

// 5. Luganda tracking: append one section (the file's last section varies by base).
{
  const rel = 'docs/i18n/pending-luganda-translations.md';
  const raw = fs.readFileSync(rel, 'utf8');
  if (!raw.startsWith('# Pending human Luganda translations\n')) fail(`${rel} is not the Luganda tracker`);
  if (raw.includes('settings.json` — Users page')) fail(`${rel} already tracks the Users keys`);
  if (!raw.endsWith('\n') || raw.endsWith('\n\n')) fail(`${rel} does not end with exactly one newline`);

  edits.set(rel, replaceOnce(fs.readFileSync(rel, 'utf8'),
